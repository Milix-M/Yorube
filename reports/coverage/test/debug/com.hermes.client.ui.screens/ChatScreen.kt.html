package com.hermes.client.ui.screens

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun ChatScreen(
    sessionId: String?,
    onOpenSettings: () -> Unit,
    onNavigateBack: () -> Unit,
) {
    val app = LocalContext.current.applicationContext as HermesApp
    val vm: ChatViewModel = viewModel(
        key = "chat-${sessionId.orEmpty()}",
        factory = object : ViewModelProvider.Factory {
            @Suppress("UNCHECKED_CAST")
            override fun <T : ViewModel> create(modelClass: Class<T>): T =
                ChatViewModel(app.container, sessionId) as T
        }
    )
    val snackbar = remember { SnackbarHostState() }
    val listState = rememberLazyListState()
    val clarifyAnswer = remember { mutableStateOf("") }

    LaunchedEffect(vm.error) {
        vm.error?.let { snackbar.showSnackbar(it); vm.dismissError() }
    }
    LaunchedEffect(vm.messages.size) {
        if (vm.messages.isNotEmpty()) listState.animateScrollToItem(vm.messages.size - 1)
    }

    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text(if (vm.sessionId.isBlank()) stringResource(R.string.new_chat) else vm.sessionId.take(20)) },
                navigationIcon = {
                    TextButton(onClick = onNavigateBack) { Text("‹") }
                },
                actions = {
                    if (vm.busy) {
                        TextButton(onClick = vm::onStop) { Text(stringResource(R.string.stop)) }
                    }
                    TextButton(onClick = onOpenSettings) { Text("⚙") }
                }
            )
        },
        snackbarHost = { SnackbarHost(snackbar) },
        bottomBar = {
            Column(Modifier.fillMaxWidth().padding(8.dp)) {
                if (vm.statusLine.isNotBlank()) {
                    Text(vm.statusLine, style = MaterialTheme.typography.labelSmall)
                }
                Row(Modifier.fillMaxWidth()) {
                    OutlinedTextField(
                        value = vm.input,
                        onValueChange = { vm.input = it },
                        modifier = Modifier.weight(1f).padding(end = 8.dp).testTag("chatInput"),
                        placeholder = {
                            Text(
                                if (vm.busy) stringResource(R.string.steer_hint)
                                else stringResource(R.string.message_hint)
                            )
                        },
                        maxLines = 4,
                    )
                    if (vm.busy) {
                        Button(onClick = vm::onSteer, modifier = Modifier.testTag("sendButton")) { Text("→") }
                    } else {
                        Button(onClick = vm::onSend, modifier = Modifier.testTag("sendButton")) { Text("↑") }
                    }
                }
            }
        }
    ) { padding ->
        LazyColumn(
            state = listState,
            modifier = Modifier.fillMaxSize().padding(padding).padding(horizontal = 8.dp),
        ) {
            items(vm.messages, key = { it.id }) { msg ->
                MessageCard(msg)
            }
        }
    }

    vm.approval?.let { req ->
        AlertDialog(
            onDismissRequest = { /* must choose */ },
            confirmButton = {},
            title = { Text(stringResource(R.string.approval_title)) },
            text = {
                Column {
                    req.command?.let { Text(it) }
                    req.description?.let { Text(it) }
                    Row {
                        for (choice in req.choices) {
                            OutlinedButton(
                                onClick = { vm.answerApproval(choice) },
                                modifier = Modifier.padding(end = 8.dp),
                            ) { Text(choice) }
                        }
                    }
                }
            },
        )
    }

    vm.clarify?.let { prompt ->
        AlertDialog(
            onDismissRequest = { /* must answer */ },
            confirmButton = {
                TextButton(onClick = {
                    vm.answerClarify(clarifyAnswer.value)
                    clarifyAnswer.value = ""
                }) { Text(stringResource(R.string.send)) }
            },
            title = { Text(stringResource(R.string.question_title)) },
            text = {
                Column {
                    Text(prompt.prompt)
                    OutlinedTextField(
                        value = clarifyAnswer.value,
                        onValueChange = { clarifyAnswer.value = it },
                        modifier = Modifier.fillMaxWidth().testTag("clarifyInput"),
                    )
                }
            },
        )
    }
}

@Composable
private fun MessageCard(msg: UiMessage) {
    val container = when (msg.kind) {
        MsgKind.USER -> MaterialTheme.colorScheme.primaryContainer
        MsgKind.ERROR -> MaterialTheme.colorScheme.errorContainer
        MsgKind.TOOL, MsgKind.STATUS -> MaterialTheme.colorScheme.surfaceVariant
        MsgKind.ASSISTANT -> MaterialTheme.colorScheme.surface
    }
    Card(
        colors = CardDefaults.cardColors(containerColor = container),
        modifier = Modifier.fillMaxWidth().padding(vertical = 4.dp),
    ) {
        Text(
            msg.text,
            style = if (msg.kind == MsgKind.TOOL || msg.kind == MsgKind.STATUS) {
                MaterialTheme.typography.labelMedium
            } else {
                MaterialTheme.typography.bodyMedium
            },
            modifier = Modifier.padding(10.dp),
        )
    }
}
